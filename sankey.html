1): parent = levels[i] child = levels[i+1] # Calculate total volume for each connection table = subjectData.groupby([parent, child])["volume"].sum() table = table.reset_index() table.columns = ["source", "target", "volume"] # Keep repeated names at different levels separate table["source"] = parent + ": " + table["source"] table["target"] = child + ": " + table["target"] tables.append(table) # Combine the connection tables connections = pd.concat(tables, ignore_index=True) connections.tail(5) # Create a list of region labels all_labels = pd.concat([connections["source"], connections["target"]]) labels = all_labels.unique().tolist() # Generate node numbers node_num = {} for number, name in enumerate(labels): node_num[name] = number # Set source, target, value source = connections["source"].map(node_num).tolist() target = connections["target"].map(node_num).tolist() value = connections["volume"].tolist() # create the sankey diagram fig = go.Figure(data=[go.Sankey( valueformat = ".0f", valuesuffix = "TWh", # Define nodes node=dict( label=labels, pad=10, thickness=15, line=dict(color="black", width=0.5), color="blue" ), # Add links link=dict( source=source, target=target, value=value ) )]) fig.update_layout( title_text="Subject 127 — Type 1: MRICloud Regional Volumes", font_size=10, width=1400, height=1600 ) fig.show()
